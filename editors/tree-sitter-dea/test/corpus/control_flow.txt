Control flow and cleanup

module flow;

enum Result {
    Value(value: int);
    Empty;
}

func inspect(result: Result?, limit: int) -> int {
    let answer = 0;
    let i = 0;

    while (i < limit) {
        if (i == 3) break;
        i = i + 1;
    }

    for (let j = 0; j < limit; j = j + 1) {
        if (j == 2) continue;
        answer = answer + j;
    }

    match (result as Result) {
        Value(value) => { answer = value; }
        Empty => { answer = 0; }
    }

    case (answer) {
        -1 => { answer = 0; }
        1 => answer = 2;
        _ => { answer = limit; }
    }

    with (let active: Result? = result) {
        answer = answer + 1;
    } cleanup {
        if (active == null) answer = 0;
    }

    return answer;
}
