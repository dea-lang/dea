type Callback = (unsafe func(int*, string...) -> int)?;
type Matrix = int[2][3];
type Window = int?*[];

func evaluate(callback: Callback, matrix: Matrix, values: int[]) -> int {
    let built = int[3]([1, 2, 3]);
    let row: int[] = slice(matrix[0], 0, 2);
    let newline: byte = '\n';
    let bell: byte = '\a';
    let value = callback?(matrix[0][0], "items", values...);
    let bits = ((value << 2) & 0xFF) | (~value ^ 0b10);
    let pointer = new int[4];
    drop pointer;
    return bits as int?;
}
